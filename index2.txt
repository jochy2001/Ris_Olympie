    <script>
        let sjrData = [];
        let currentIndex = 0;

        // 1. Hodiny
        setInterval(() => {
            const now = new Date();
            document.getElementById("clock").innerText = now.toLocaleTimeString('cs-CZ');
            updateDelay();
        }, 1000);

        // 2. Načtení CSV z Google Disku
        async function loadSJR() {
            const url = document.getElementById("csv-url").value;
            if (!url) return alert("Vlož URL odkaz na CSV!");

            try {
                const response = await fetch(url);
                const text = await response.text();
                parseCSV(text);
            } catch (err) {
                alert("Chyba při načítání. Zkontroluj odkaz a veřejné sdílení souboru.");
            }
        }

        // 3. Zpracování CSV
        function parseCSV(text) {
            const lines = text.split("\n");
            sjrData = [];

            for (let i = 1; i < lines.length; i++) {
                const cols = lines[i].split(",");
                if (cols.length >= 3) {
                    sjrData.push({
                        station: cols[0].trim(),
                        depTime: cols[1].trim(),
                        maxSpeed: cols[2].trim()
                    });
                }
            }

            if (sjrData.length > 0) {
                currentIndex = 0;
                renderTable();
                selectStation(0);
            }
        }

        // 4. Vykreslení tabulky
        function renderTable() {
            let tableHtml = "";
            sjrData.forEach((item, index) => {
                const rowClass = index === currentIndex ? "active-row" : "";
                tableHtml += `<tr class="${rowClass}">
                    <td>${item.station}</td>
                    <td>${item.depTime}</td>
                    <td>${item.maxSpeed} km/h</td>
                </tr>`;
            });
            document.getElementById("sjr-table").innerHTML = tableHtml;
        }

        // 5. Výběr stanice
        function selectStation(index) {
            if (index >= sjrData.length) return;
            
            currentIndex = index;
            const item = sjrData[currentIndex];
            
            document.getElementById("current-station").innerText = item.station;
            document.getElementById("schedule-info").innerText = `Odjezd dle SJŘ: ${item.depTime} | Max: ${item.maxSpeed} km/h`;
            
            renderTable();
            updateDelay();
        }

        // 6. Tlačítko Další stanice
        function nextStation() {
            if (currentIndex < sjrData.length - 1) {
                selectStation(currentIndex + 1);
            } else {
                alert("Jsi na konci jízdního řádu.");
            }
        }

        // 7. Výpočet zpoždění oproti hodinám
        function updateDelay() {
            if (sjrData.length === 0) return;

            const item = sjrData[currentIndex];
            if (!item.depTime.includes(":")) return;

            const [h, m] = item.depTime.split(":").map(Number);
            const now = new Date();
            
            const plannedTime = new Date();
            plannedTime.setHours(h, m, 0, 0);

            // Rozdíl v minutách
            const diffMs = now - plannedTime;
            const diffMinutes = Math.floor(diffMs / 60000);

            const statusElem = document.getElementById("status-delay");
            if (diffMinutes > 0) {
                statusElem.innerText = `Zpoždění: +${diffMinutes} min`;
                statusElem.className = "delay-text delay-late";
            } else if (diffMinutes < 0) {
                statusElem.innerText = `Předstih: ${Math.abs(diffMinutes)} min`;
                statusElem.className = "delay-text delay-ok";
            } else {
                statusElem.innerText = "Jedeš přesně podle řádu";
                statusElem.className = "delay-text delay-ok";
            }
        }
    </script>
